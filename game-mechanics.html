النظام الكيميائي في BioPulse

نظام الإشارات الكيميائية المتقدم

في BioPulse، يمثل النظام الكيميائي قلب المحاكاة البيولوجية، حيث يتحكم في تفاعل الكائنات الحية مع بيئتها من خلال إفراز واستقبال الإشارات الكيميائية.

يعتمد النظام على مفهوم المركبات الكيميائية (Chemical Compounds) التي تحمل معلومات وتأثيرات محددة، وتنتشر في البيئة وفق نماذج رياضية واقعية.

💡 شرح مبسط للنظام الكيميائي

تخيل أن الكائنات الحية في اللعبة تتواصل مع بعضها باستخدام "رسائل كيميائية" تنتشر في البيئة المحيطة.

كل رسالة كيميائية لها:

  • نوع: مثل "سم" أو "مضاد حيوي" أو "مادة مغذية"
  • قوة: تحدد مدى تأثيرها ومدى انتشارها
  • تأثير: مثل إبطاء الحركة أو زيادة الصحة أو منع التكاثر

void CreateChemicalSignal(ChemicalCompound compound, float intensity, float Range = 0)
{
                        GameObject signalObj = new GameObject($"ChemicalSignal_{compound.name}_{Time.time}");
    signalObj.transform.position = transform.position;

                        int chemicalLayer = LayerMask.NameToLayer("ChemicalSignals");

                        if (chemicalLayer != -1)
    {
        signalObj.layer = chemicalLayer;
    }

                        EnhancedChemicalSignal signal = signalObj.AddComponent<EnhancedChemicalSignal>();

                        // حساب المدى بناءً على الشدة
                        float signalRange = Mathf.Lerp(minSecretionRange, maxSecretionRange, intensity / compound.maxIntensity);
    signalRange = Mathf.Clamp(signalRange, compound.baseRadius, compound.maxRadius);

                        // تمرير المصدر الصحيح
                        if(Range > 0)
    {
        signal.Initialize(compound, intensity, parentEntity, Range, Tus);
    }
                        else
    {
        signal.Initialize(compound, intensity, parentEntity, signalRange, Tus);
    }
    InteractWithExistingSignals(signal);
}

المعاملات الأساسية للنظام الكيميائي

كل معامل في الدالة السابقة يلعب دوراً محورياً في تحديد سلوك الإشارة الكيميائية وطريقة تفاعلها مع البيئة المحيطة.

ChemicalCompound compound

المركب الكيميائي الذي يحمل جميع الخصائص والتأثيرات.

float intensity

شدة الإشارة الكيميائية، تؤثر على قوة التأثير ومدى الانتشار.

parentEntity

الكائن المصدر للإشارة، يمنع الكائن من الاستجابة لإشاراته الخاصة.

signalRange

المدى المحسوب تلقائياً بناءً على الشدة وخصائص المركب.

Range

المدى المحدد يدوياً، يستخدم لتجاوز الحساب التلقائي.

Tus

نظام الأنسجة النسيجية الذي يتحكم في تفاعل الإشارات مع الأنسجة.

النماذج الحسابية لانتشار الإشارات

مقارنة نماذج انتشار الإشارات الكيميائية

النموذج الخطي

I = I₀ × (1 - d/R)

المميزات

  • بسيط وسهل الفهم
  • أداء حسابي عالي

العيوب

  • غير واقعي للأنظمة البيولوجية
  • تغير مفاجئ عند المسافة R

النموذج الأسي

I = I₀ × e^(-kd)

المميزات

  • واقعي بيولوجيًا
  • تناقص تدريجي طبيعي

العيوب

  • أكثر تعقيدًا
  • يتطلب معايرة دقيقة

النموذج التربيعي العكسي

I = I₀ / d²

المميزات

  • دقيق فيزيائيًا للإشعاع
  • بسيط حسابيًا

العيوب

  • غير واقعي للأنسجة البيولوجية
  • تناقص سريع جدًا

🧪 النموذج الغاوسي - الخيار المتقدم

I = I₀ × e^(-d²/2σ²) – واقعي جدًا للأنظمة البيولوجية ولكن بتكلفة حسابية أعلى، يمكن تحسينه باستخدام Caching.

Genome: Viral Evolution – آليات التطور الجيني

نظام الجينوم والطفرات

في Genome: Viral Evolution، تتحكم في فيروس يتطور جينومه باستمرار. كل فيروس له جينوم (Genome) يحتوي على تسلسل من القواعد النيتروجينية (A, T, C, G, U) وخصائص مثل سرعة النسخ، الاستقرار، ومعدل الطفرة. مع تراكم الخبرة (XP)، يرتفع مستوى الفيروس وتظهر بطاقات طفرات تسمح لك باختيار تطورات جديدة.

🧬 كيف يعمل نظام XP والمستويات؟

يكسب الفيروس XP عند إصابة الخلايا أو عند إكمال دورات تكاثر. عندما يصل XP إلى الحد الأقصى (maxXP)، يزيد الحد الأقصى بمقدار 100 ويتم عرض 3 بطاقات طفرات للاختيار من بينها. يمكن أن تكون الطفرات:

  • ReplicationSpeedMutation: يغير سرعة النسخ الأساسية (replicationBaseSpeed).
  • ProteinMutation: يغير قوة (potency) أحد البروتينات المنتجة.
  • GenomeLengthMutation: يزيد أو يقلل طول الجينوم (GenomeLength) ضمن حدود 5-50، مع تحديث التسلسل وتأثيرات على العتبات (Thresholds).
Genome.cs – جزء من نظام XP والطفرات
public void IncreaseXP(int amount)
{
    XP = Mathf.Min(XP + amount, maxXP);
    OnXPChanged?.Invoke(XP, maxXP);
    ApplyRandomMinorMutation();

                        if (XP >= maxXP)
    {
        XP = 0;
        LevelUp();
    }
}

private void LevelUp()
{
    maxXP += 100;
    XP = 0;
    OnXPChanged?.Invoke(XP, maxXP);
    CardUiManger.Instance.ShowLevelUpCards(this, 3);
}

private void ApplyRandomMinorMutation()
{
    stack++;
                        float chance = Mathf.Min(stack * 0.1f, 0.6f);
    Mutation minorMutation;

                        if (UnityEngine.Random.value < chance)
    {
        minorMutation = new GenomeLengthMutation();
        stack = 0;
        requiredStackForLength += 2;
    }
                        else
    {
                        int randomType = UnityEngine.Random.Range(0, 2);
                        if (randomType == 0)
            minorMutation = new ProteinMutation();
                        else
            minorMutation = new ReplicationSpeedMutation();
    }

    ApplyMutation(minorMutation);

                        if (minorMutation is GenomeLengthMutation && CheckAndUpdateGenomeThreshold())
    {
        CardUiManger.Instance.ShowLevelUpCardOnSpawns(this, 1);
    }
}

🔀 شرح الطفرات الرئيسية

  • ReplicationSpeedMutation: تغيير عشوائي بين -0.15 و +0.25 في سرعة النسخ، مع تحديد حد أدنى 0.1 وحد أقصى 5.
  • ProteinMutation: اختيار بروتين عشوائي من قائمة ProteinsProdust وتغيير قوته (potency) إلى قيمة عشوائية بين 0 و 4.
  • GenomeLengthMutation: زيادة أو إنقاص طول الجينوم بمقدار 1 أو 2 (مع مقاومة للزيادة عند الطول الكبير)، ثم مزامنة التسلسل وإعادة حساب العتبة (GenomeLengthThreshold).

عتبات طول الجينوم (GenomeLengthThreshold)

كلما زاد طول الجينوم، يصل الفيروس إلى مراحل تطورية جديدة تعكس تعقيده:

  • VeryLow (2) – بداية ضعيفة
  • Low (5)
  • Medium (10)
  • High (15)
  • Extreme (20)
  • Ultra (30)
  • Max (50) – أقصى تعقيد

عند الوصول إلى عتبة جديدة، يتم إطلاق حدث (OnThresholdChanged) ويمكن عرض بطاقة تطور إضافية.

Genome.cs – حساب العتبة
public bool CheckAndUpdateGenomeThreshold()
{
    GenomeLengthThreshold newThreshold = genomeLengthThreshold;

                        if (GenomeLength >= 20)
        newThreshold = GenomeLengthThreshold.Max;
                        else if (GenomeLength >= 18)
        newThreshold = GenomeLengthThreshold.Ultra;
                        else if (GenomeLength >= 15)
        newThreshold = GenomeLengthThreshold.Extreme;
                        else if (GenomeLength >= 13)
        newThreshold = GenomeLengthThreshold.High;
                        else if (GenomeLength >= 10)
        newThreshold = GenomeLengthThreshold.Medium;
                        else if (GenomeLength >= 7)
        newThreshold = GenomeLengthThreshold.Low;
                        else
        newThreshold = GenomeLengthThreshold.VeryLow;

                        if (newThreshold != genomeLengthThreshold)
    {
        genomeLengthThreshold = newThreshold;
        OnThresholdChanged?.Invoke(genomeLengthThreshold);
                        return true;
    }
                        return false;
}

نظام الشبكة (Grid) والاقتراب

يستخدم العالم في اللعبة نظام شبكي (Grid) لتحديد المواقع والاقتراب بين الكائنات. كل خلية وفيروس هو IGridEntity، ويتم تخزينهم في Dictionary<Vector2Int, List<IGridEntity>>. الخلايا والفيروسات يمكنها البحث عن الكائنات القريبة ضمن نطاق معين.

GridManager.cs – البحث عن الكائنات القريبة
public List<IGridEntity> GetNearby(Vector2 position, int range = 1)
{
    List<IGridEntity> result = new List<IGridEntity>();
    Vector2Int center = GetCell(position);

                        for (int x = -range; x <= range; x++)
    {
                        for (int y = -range; y <= range; y++)
        {
            Vector2Int check = center + new Vector2Int(x, y);
                        if (grid.ContainsKey(check))
                result.AddRange(grid[check]);
        }
    }
                        return result;
}

دورة حياة العدوى

عندما يلامس الفيروس خلية سليمة، تبدأ عملية العدوى:

  • يتم إنشاء حويصلة (Vesicle) لهضم الفيروس.
  • تصبح الخلية مصابة (infected) وتكتسب جينوم الفيروس.
  • تبدأ الخلية في إنتاج فيروسات جديدة داخلياً (internalViruses) أو عن طريق الانفجار (Burst).
  • عند وصول عدد الفيروسات الداخلية إلى maxInternalViruses، تنفجر الخلية وتطلق الفيروسات والبروتينات (Organelles) في البيئة.
Cell.cs – التعامل مع ملامسة الفيروس
public void HandleVirusContact(Virus virus)
{
                        if (vesiclePrefab != null && vesicleSpawnPoint != null && vesicleEXITPoint != null)
    {
        GameObject vesObj = Instantiate(vesiclePrefab, transform.position, Quaternion.identity);
        Vesicle vesicle = vesObj.GetComponent<Vesicle>();
        vesicle.Init(virus, this, vesicleSpawnPoint, vesicleEXITPoint);
        vesicle.DigestVirus();
    }

                        if (!infected)
    {
        infected = true;
        genome = virus.genome;
        StartCoroutine(FlashRed());
        StartCoroutine(InfectedLightPulse());
                        if (infectionParticles != null) infectionParticles.Play();
        SetInfectedActivity();
        ApplyInfectionShader();

                        if (!startedInfecting)
        {
            startedInfecting = true;
            StartCoroutine(ProduceViruses());
        }
    }
}

نظام الدفاع المناعي

الخلايا السليمة تدافع عن نفسها عن طريق إطلاق قذائف مناعية (ImmuneProjectile) تستهدف الفيروسات القريبة. سرعة الإطلاق (defenseFireRate) وعدد الطلقات (shotsPerBurst) ومدى الدفاع (defenseRadius) قابلة للتعديل.

Cell.cs – الدفاع المناعي
IEnumerator ImmuneBurst(Virus target)
{
                        if (target == null) yield break;
                        if (organelles == null || organelles.Length == 0) yield break;
                        if (immuneProjectilePrefab == null) yield break;

                        if (defensePulse != null) defensePulse.Play();
                        if (defenseLight != null) defenseLight.intensity = 2f;

                        int shots = Mathf.Min(shotsPerBurst, organelles.Length);

                        for (int i = 0; i < shots; i++)
    {
                        if (organelles[i] == null) continue;
        Transform origin = organelles[i].transform;
        Vector2 dir = (Vector2)target.transform.position - (Vector2)origin.position;
                        float angle = Mathf.Atan2(dir.y, dir.x) * Mathf.Rad2Deg;
        GameObject obj = Instantiate(immuneProjectilePrefab, origin.position, Quaternion.Euler(0, 0, angle));
        ImmuneProjectile proj = obj.GetComponent<ImmuneProjectile>();
        VirusCore core = target.GetComponentInChildren<VirusCore>();
                        if (proj != null) proj.Init(core);
        yield return new WaitForSeconds(0.08f);
    }

                        if (defenseLight != null) defenseLight.intensity = baseIntensity;
                        if (target != null) target.targetedBy.Remove(this);
}

تأثيرات الضرر على الفيروسات والخلايا

كل فيروس يتأثر بالضرر من خلال فلاش أبيض واهتزاز موضعي (Hit Shake). الخلايا المصابة تومض باللون الأحمر وتنبض بالضوء. عند انفجار الخلية، تظهر جسيمات انفجار وموجة صدمية (Shockwave).

Virus.cs – تطبيق الضرر مع التأثيرات
public void TakeDamage(float damage)
{
    currentHealth -= damage;
    VirusVFX.FlashWhite(this, spriteRenderer, duration: 0.05f);
    VirusVFX.HitShake(this, transform, amplitude: 0.05f, duration: 0.1f);
                        if (currentHealth <= 0) Die();
}
